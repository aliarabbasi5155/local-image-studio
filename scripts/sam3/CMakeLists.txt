cmake_minimum_required(VERSION 3.14)
project(studio_sam3 LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release)
set(SAM3_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SAM3_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../runtime/sam3.cpp" sam3)
add_executable(studio-sam3 main.cpp)
target_link_libraries(studio-sam3 PRIVATE sam3)
set_target_properties(studio-sam3 PROPERTIES CXX_STANDARD 14)
